**Fixes made to the handout code (important):**

- Ex 3: In Lex, the `{ ... }` action must start on the same line as its pattern. The handout puts `{` on the next line, which gives a flex error. Fixed.
- Ex 3: The handout's expected counts (4, 7, 3, 1, 1, 10, 1) are wrong for the given input. The correct counts are 5, 8, 3, 1, 1, 12, 1 (`int main() { int a = 10; int b = 20; if(a < b) { a = a + b; } return 0; }` has 5 keywords: int, int, int, if, return).
- Ex 3: The `/` arithmetic rule is placed after the comment rules so `//` and `/* */` are not read as division.
- Ex 8: `ID` must be declared as `%token <str> ID`, otherwise bison errors with '$2 of declaration has no declared type'.
- Ex 8 and 9: The lexer includes the header generated by bison (`typecheck.tab.h` / `balance.tab.h`). The handout uses `y.tab.h`, which only exists with the old `yacc -d` command. Use the bison commands below.

## Lab Exercise 1: Lexical Analyzer (in C)

**File:** `lexer.c`

```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>

#define NKEY 8
const char *keywords[NKEY] = {"int", "float", "char", "double", "if", "else", "return", "while"};

int isKeyword(const char *s)
{
    for (int i = 0; i < NKEY; i++)
        if (strcmp(s, keywords[i]) == 0)
            return 1;
    return 0;
}

int main(void)
{
    const char *src =
        "int main() {\n"
        "    int a = 25;\n"
        "    float b = 10.5;\n"
        "    if (a >= 20) {\n"
        "        b = b + a;\n"
        "    }\n"
        "    return 0;\n"
        "}\n";

    int i = 0, n = strlen(src);
    char buf[64];

    printf("Token Stream\n\n");

    while (i < n)
    {
        char c = src[i];

        if (isspace((unsigned char)c)) { i++; continue; }

        /* identifier or keyword */
        if (isalpha((unsigned char)c) || c == '_')
        {
            int k = 0;
            while (isalnum((unsigned char)src[i]) || src[i] == '_')
                buf[k++] = src[i++];
            buf[k] = '\0';
            printf("<%s, %s>\n", buf, isKeyword(buf) ? "KEYWORD" : "IDENTIFIER");
            continue;
        }

        /* integer or floating-point constant */
        if (isdigit((unsigned char)c))
        {
            int k = 0, isFloat = 0;
            while (isdigit((unsigned char)src[i]) || (src[i] == '.' && !isFloat))
            {
                if (src[i] == '.') isFloat = 1;
                buf[k++] = src[i++];
            }
            buf[k] = '\0';
            printf("<%s, %s>\n", buf, isFloat ? "FLOAT_CONSTANT" : "INTEGER_CONSTANT");
            continue;
        }

        /* operators */
        if ((c == '>' || c == '<' || c == '=' || c == '!') && src[i + 1] == '=')
        {
            const char *t = (c == '=') ? "RELATIONAL_OPERATOR" : "RELATIONAL_OPERATOR";
            printf("<%c=, %s>\n", c, t);
            i += 2;
            continue;
        }
        if (c == '>' || c == '<') { printf("<%c, RELATIONAL_OPERATOR>\n", c); i++; continue; }
        if (c == '=') { printf("<=, ASSIGNMENT_OPERATOR>\n"); i++; continue; }
        if (strchr("+-*/%", c)) { printf("<%c, ARITHMETIC_OPERATOR>\n", c); i++; continue; }

        /* separators, parentheses, braces */
        if (c == ';' || c == ',') { printf("<%c, SEPARATOR>\n", c); i++; continue; }
        if (c == '(') { printf("<(, LEFT_PARENTHESIS>\n"); i++; continue; }
        if (c == ')') { printf("<), RIGHT_PARENTHESIS>\n"); i++; continue; }
        if (c == '{') { printf("<{, LEFT_BRACE>\n"); i++; continue; }
        if (c == '}') { printf("<}, RIGHT_BRACE>\n"); i++; continue; }

        printf("<%c, UNKNOWN>\n", c);
        i++;
    }
    return 0;
}
```

## Lab Exercise 2: Symbol Table in C

**Files:** `symtab.c`, `input.c`

```c
#include <stdio.h>
#include <string.h>

#define MAX 100

struct Symbol {
    char name[32];
    char type[10];
    int size;
    int address;
    char scope[10];
};

int getSize(const char *type)
{
    if (strcmp(type, "int") == 0)    return 4;
    if (strcmp(type, "float") == 0)  return 4;
    if (strcmp(type, "char") == 0)   return 1;
    if (strcmp(type, "double") == 0) return 8;
    return 0;
}

int main(void)
{
    struct Symbol table[MAX];
    int count = 0, address = 1000;
    char type[20], name[40], line[100];

    printf("Enter declarations (one per line, e.g. int x;). Ctrl+D to finish:\n");

    while (fgets(line, sizeof line, stdin))
    {
        if (sscanf(line, "%19s %39[^;=\n ]", type, name) != 2)
            continue;

        int size = getSize(type);
        if (size == 0)
        {
            printf("Skipping unsupported type: %s\n", type);
            continue;
        }

        strcpy(table[count].name, name);
        strcpy(table[count].type, type);
        table[count].size = size;
        table[count].address = address;
        strcpy(table[count].scope, "Global");

        address += size;
        count++;
    }

    printf("\n%-12s %-10s %-14s %-8s %-6s\n",
           "Identifier", "Data Type", "Size (Bytes)", "Address", "Scope");
    printf("--------------------------------------------------------\n");
    for (int i = 0; i < count; i++)
        printf("%-12s %-10s %-14d %-8d %-6s\n",
               table[i].name, table[i].type, table[i].size,
               table[i].address, table[i].scope);

    return 0;
}
```

## Lab Exercise 5: FIRST and FOLLOW

**File:** `first_follow.c`

The grammar is hard-coded using single characters (`X` stands for E', `Y` for T', `e` for epsilon, `i` for id) so that every symbol is one character. To test another grammar, edit the `g[]` array in `main()`, set `np`, and the `nts[]` list.

```c
#include <stdio.h>
#include <string.h>
#include <ctype.h>

/* Grammar is stored one production per entry: lhs + rhs string.
   Each symbol is a single character. Use 'e' for epsilon.
   Non-terminals are uppercase letters. E' is written as X, T' as Y here is
   handled by a rename table so the output prints E' and T'. */

#define MAXP 20
#define MAXS 20

char lhs[MAXP];
char rhs[MAXP][20];
int np = 0;

char first[128][MAXS];
char follow[128][MAXS];
char startSym;

int addTo(char set[128][MAXS], char nt, char c)
{
    char *s = set[(int)nt];
    if (strchr(s, c)) return 0;
    int n = strlen(s);
    s[n] = c;
    s[n + 1] = '\0';
    return 1;
}

int isNonTerminal(char c) { return isupper((unsigned char)c); }

void computeFirst(void)
{
    int changed = 1;
    while (changed)
    {
        changed = 0;
        for (int p = 0; p < np; p++)
        {
            char A = lhs[p];
            char *r = rhs[p];
            int allEps = 1;

            for (int i = 0; r[i]; i++)
            {
                char X = r[i];
                if (!isNonTerminal(X))
                {
                    if (X == 'e') { changed |= addTo(first, A, 'e'); }
                    else          { changed |= addTo(first, A, X); allEps = 0; }
                    if (X != 'e') break;
                    continue;
                }
                for (char *f = first[(int)X]; *f; f++)
                    if (*f != 'e')
                        changed |= addTo(first, A, *f);

                if (!strchr(first[(int)X], 'e')) { allEps = 0; break; }
            }
            if (allEps) changed |= addTo(first, A, 'e');
        }
    }
}

/* FIRST of a string beta; result in out, includes 'e' if beta can vanish */
void firstOfString(const char *beta, char *out)
{
    out[0] = '\0';
    int allEps = 1;
    for (int i = 0; beta[i]; i++)
    {
        char X = beta[i];
        if (!isNonTerminal(X))
        {
            if (X != 'e')
            {
                if (!strchr(out, X)) { int n = strlen(out); out[n] = X; out[n + 1] = 0; }
                allEps = 0;
                break;
            }
            continue;
        }
        for (char *f = first[(int)X]; *f; f++)
            if (*f != 'e' && !strchr(out, *f)) { int n = strlen(out); out[n] = *f; out[n + 1] = 0; }
        if (!strchr(first[(int)X], 'e')) { allEps = 0; break; }
    }
    if (allEps && !strchr(out, 'e')) { int n = strlen(out); out[n] = 'e'; out[n + 1] = 0; }
}

void computeFollow(void)
{
    addTo(follow, startSym, '$');
    int changed = 1;
    while (changed)
    {
        changed = 0;
        for (int p = 0; p < np; p++)
        {
            char A = lhs[p];
            char *r = rhs[p];
            for (int i = 0; r[i]; i++)
            {
                char B = r[i];
                if (!isNonTerminal(B)) continue;

                char fb[MAXS];
                firstOfString(r + i + 1, fb);

                for (char *f = fb; *f; f++)
                    if (*f != 'e')
                        changed |= addTo(follow, B, *f);

                if (r[i + 1] == '\0' || strchr(fb, 'e'))
                    for (char *f = follow[(int)A]; *f; f++)
                        changed |= addTo(follow, B, *f);
            }
        }
    }
}

/* pretty-print names: X -> E', Y -> T' */
void printName(char c)
{
    if (c == 'X') printf("E'");
    else if (c == 'Y') printf("T'");
    else printf("%c", c);
}

void printSet(char *s)
{
    printf("{ ");
    for (int i = 0; s[i]; i++)
    {
        if (i) printf(", ");
        if (s[i] == 'e') printf("ε");
        else if (s[i] == 'i') printf("id");
        else printf("%c", s[i]);
    }
    printf(" }\n");
}

int main(void)
{
    /* E->TX, X->+TX|e, T->FY, Y->*FY|e, F->(E)|i   (i = id) */
    const char *g[] = {"E=TX", "X=+TX", "X=e", "T=FY", "Y=*FY", "Y=e", "F=(E)", "F=i"};
    np = 8;
    for (int i = 0; i < np; i++)
    {
        lhs[i] = g[i][0];
        strcpy(rhs[i], g[i] + 2);
    }
    startSym = 'E';

    printf("Grammar:\n");
    printf("E  -> T E'\nE' -> + T E' | ε\nT  -> F T'\nT' -> * F T' | ε\nF  -> ( E ) | id\n\n");

    computeFirst();
    computeFollow();

    char nts[] = {'E', 'X', 'T', 'Y', 'F'};

    printf("FIRST Sets\n");
    for (int i = 0; i < 5; i++)
    {
        printf("FIRST(");  printName(nts[i]);  printf(") = ");
        printSet(first[(int)nts[i]]);
    }

    printf("\nFOLLOW Sets\n");
    for (int i = 0; i < 5; i++)
    {
        printf("FOLLOW("); printName(nts[i]); printf(") = ");
        printSet(follow[(int)nts[i]]);
    }
    return 0;
}
```

## Lab Exercise 6: Shift-Reduce Parser

**File:** `shift_reduce.c`

```c
#include <stdio.h>
#include <string.h>

#define MAXP 20

char lhs[MAXP];
char rhs[MAXP][30];
int n;

char stack[200];
char input[200];

/* try to reduce top of stack using any production; returns production index or -1 */
int tryReduce(void)
{
    int sl = strlen(stack);
    for (int p = 0; p < n; p++)
    {
        int rl = strlen(rhs[p]);
        if (rl <= sl && strcmp(stack + sl - rl, rhs[p]) == 0)
            return p;
    }
    return -1;
}

int main(void)
{
    char str[100];

    printf("Enter the number of productions:\n");
    scanf("%d", &n);

    printf("\nEnter the productions (format E->E+E):\n");
    for (int i = 0; i < n; i++)
    {
        char prod[60];
        scanf("%59s", prod);
        lhs[i] = prod[0];
        strcpy(rhs[i], prod + 3);
    }

    printf("\nEnter the input string:\n");
    scanf("%99s", str);

    printf("\nGrammar:\n");
    for (int i = 0; i < n; i++)
        printf("%c->%s\n", lhs[i], rhs[i]);
    printf("\nInput String: %s\n\n", str);

    strcpy(input, str);
    strcat(input, "$");
    stack[0] = '\0';
    int ip = 0;
    char startSym = lhs[0];

    printf("---------------------------------------------------------\n");
    printf("%-16s %-16s %s\n", "Stack", "Input", "Action");
    printf("---------------------------------------------------------\n");

    int accepted = 0;

    while (1)
    {
        /* reduce whenever possible */
        int p;
        while ((p = tryReduce()) != -1)
        {
            /* leave a pending shift if we could still extend, handled by greedy rule:
               reduce immediately only when it is the final handle or id-like rule */
            printf("%-16s %-16s Reduce %c->%s\n", stack, input + ip, lhs[p], rhs[p]);
            int sl = strlen(stack) - strlen(rhs[p]);
            stack[sl] = lhs[p];
            stack[sl + 1] = '\0';
        }

        if (input[ip] == '$')
        {
            if (strlen(stack) == 1 && stack[0] == startSym)
            {
                printf("%-16s %-16s Accept\n", stack, input + ip);
                accepted = 1;
            }
            break;
        }

        /* shift: id is treated as one symbol, handled by shifting "id" together */
        printf("%-16s %-16s Shift\n", stack, input + ip);
        if (input[ip] == 'i' && input[ip + 1] == 'd')
        {
            strcat(stack, "id");
            ip += 2;
        }
        else
        {
            int sl = strlen(stack);
            stack[sl] = input[ip++];
            stack[sl + 1] = '\0';
        }
    }

    printf("---------------------------------------------------------\n");
    printf("Result: String %s\n", accepted ? "Accepted" : "Rejected");
    return 0;
}
```

```text
E+*id            $                Reduce E->id
---------------------------------------------------------
Result: String Rejected
```

Note: the parser reduces as soon as a handle is on top of the stack. This is enough for the lab grammar, but it does not apply operator precedence (`*` over `+`).

## Lab Exercise 7: Study of LLVM

### 1. Aim
To study the LLVM compiler infrastructure, its architecture, components and tools, and to generate and execute LLVM Intermediate Representation (IR) for a simple C program.

### 2. Introduction to LLVM
LLVM (originally "Low Level Virtual Machine") is a collection of modular and reusable compiler and toolchain technologies. It was started at the University of Illinois by Chris Lattner and Vikram Adve. Its central idea is a language-independent intermediate representation (IR) that sits between source languages and machine targets. Many front ends (C/C++ via Clang, Rust, Swift, Julia and others) can generate LLVM IR, and one shared optimizer and set of back ends turn it into code for many CPUs (x86, ARM, RISC-V and more). This avoids writing a separate compiler for every language-target pair.

### 3. LLVM Architecture
LLVM follows a three-phase design:

```
 Source code  ->  [ Front End ]  ->  LLVM IR  ->  [ Optimizer ]  ->  LLVM IR  ->  [ Back End ]  ->  Machine code
 (C, C++ ...)      (Clang)                         (opt passes)                      (llc / codegen)   (x86, ARM ...)
```

With N languages and M targets, only N front ends + M back ends are needed instead of N x M compilers.

### 4. LLVM Components
- **Front end:** Reads source code, performs lexical analysis, parsing and semantic analysis, and produces LLVM IR. Clang is the front end for C, C++ and Objective-C.
- **LLVM IR:** The common, typed, low-level, SSA-form (Static Single Assignment) representation that links all phases. It exists in three equivalent forms: human-readable text (`.ll`), compact bitcode (`.bc`), and in-memory data structures.
- **Optimizer:** A sequence of passes that transform IR to IR without changing behavior: constant folding, dead code elimination, inlining, mem2reg, loop optimizations, and so on. Run with `opt`.
- **Back end:** Converts optimized IR to target machine code: instruction selection, register allocation, instruction scheduling and emission of assembly or object code. Run with `llc`.

### 5. LLVM Intermediate Representation
- **Basic structure:** A *module* contains global variables, function declarations and function definitions. Each *function* contains *basic blocks*, and each block contains *instructions*.
- **Instructions:** Typed, three-address style. Examples: `alloca` (allocate stack memory), `load`, `store`, `add`, `sub`, `mul`, `icmp` (compare), `br` (branch), `call`, `ret`.
- **Registers:** Unlimited virtual registers. Local ones are written `%name` or `%0, %1, ...`; globals and functions are written `@name`. In SSA form, each register is assigned exactly once.
- **Basic blocks:** A straight-line sequence of instructions with one entry point at the top and ending in exactly one terminator (`ret` or `br`). Unnamed blocks get numbers; control flow moves between blocks with branches.

Example of types and instructions: `%7 = add nsw i32 %5, %6` adds two 32-bit integers (`i32`) and stores the result in virtual register `%7`.

### 6. LLVM Compilation Process
`Source code -> LLVM IR -> Optimization -> Machine code`

1. Clang parses the C source and emits unoptimized IR (`clang -S -emit-llvm`).
2. `opt` runs optimization passes over the IR (`opt -O2`).
3. `llc` lowers optimized IR to assembly for the target CPU.
4. The assembler and linker produce the executable (clang does this automatically).

### 7. LLVM Tools
| Tool | Purpose |
|---|---|
| `clang` | C/C++ compiler driver and front end; can emit IR (`-S -emit-llvm`), bitcode (`-c -emit-llvm`) or executables |
| `llvm-as` | Assembles text IR (`.ll`) into bitcode (`.bc`) |
| `llvm-dis` | Disassembles bitcode (`.bc`) back into readable IR (`.ll`) |
| `opt` | Runs optimization and analysis passes on IR |
| `llc` | Compiles IR/bitcode to target assembly or object code |
| `lli` | Interpreter / JIT that directly executes bitcode or IR |

### 8. Simple Example
**C program (`sample.c`):**

```c
#include <stdio.h>

int add(int a, int b)
{
    return a + b;
}

int main()
{
    int x = 5, y = 3;
    int sum = add(x, y);
    printf("Sum = %d\n", sum);
    return 0;
}
```

**Corresponding LLVM IR (unoptimized, `sample.ll`, attribute and metadata lines omitted):**

```llvm
; ModuleID = 'sample.c'
source_filename = "sample.c"
target datalayout = "e-m:e-p270:32:32-p271:32:32-p272:64:64-i64:64-i128:128-f80:128-n8:16:32:64-S128"
target triple = "x86_64-pc-linux-gnu"

@.str = private unnamed_addr constant [10 x i8] c"Sum = %d\0A\00", align 1

define dso_local i32 @add(i32 noundef %0, i32 noundef %1) #0 {
  %3 = alloca i32, align 4
  %4 = alloca i32, align 4
  store i32 %0, ptr %3, align 4
  store i32 %1, ptr %4, align 4
  %5 = load i32, ptr %3, align 4
  %6 = load i32, ptr %4, align 4
  %7 = add nsw i32 %5, %6
  ret i32 %7
}

define dso_local i32 @main() #0 {
  %1 = alloca i32, align 4
  %2 = alloca i32, align 4
  %3 = alloca i32, align 4
  %4 = alloca i32, align 4
  store i32 0, ptr %1, align 4
  store i32 5, ptr %2, align 4
  store i32 3, ptr %3, align 4
  %5 = load i32, ptr %2, align 4
  %6 = load i32, ptr %3, align 4
  %7 = call i32 @add(i32 noundef %5, i32 noundef %6)
  store i32 %7, ptr %4, align 4
  %8 = load i32, ptr %4, align 4
  %9 = call i32 (ptr, ...) @printf(ptr noundef @.str, i32 noundef %8)
  ret i32 0
}

declare i32 @printf(ptr noundef, ...) #1
```

Reading it: `alloca` reserves stack slots for the parameters and locals, `store`/`load` move values in and out of them, `add nsw i32` performs the addition, `call` invokes `@add` and `@printf`, and `ret` returns from the function.

**After optimization (`opt -O2`):** the stack slots are removed and `add` becomes a single instruction:

```llvm
; ModuleID = 'sample.ll'
source_filename = "sample.c"
target datalayout = "e-m:e-p270:32:32-p271:32:32-p272:64:64-i64:64-i128:128-f80:128-n8:16:32:64-S128"
target triple = "x86_64-pc-linux-gnu"

@.str = private unnamed_addr constant [10 x i8] c"Sum = %d\0A\00", align 1

define dso_local i32 @add(i32 noundef %0, i32 noundef %1) local_unnamed_addr #0 {
  %3 = add nsw i32 %1, %0
  ret i32 %3
}

define dso_local noundef i32 @main() local_unnamed_addr #1 {
  %1 = tail call i32 @add(i32 noundef 5, i32 noundef 3)
  %2 = tail call i32 (ptr, ...) @printf(ptr noundef nonnull dereferenceable(1) @.str, i32 noundef %1)
  ret i32 0
}

declare noundef i32 @printf(ptr nocapture noundef readonly, ...) local_unnamed_addr #2
```

### 9. Procedure / Steps to Generate and Execute LLVM IR

```bash
# 1. Create the C file (sample.c) as shown above
# 2. Generate textual LLVM IR
clang -S -emit-llvm -O0 -Xclang -disable-O0-optnone sample.c -o sample.ll

# 3. Assemble IR to bitcode, and disassemble back
llvm-as sample.ll -o sample.bc
llvm-dis sample.bc -o sample_dis.ll

# 4. Optimize the IR
opt -O2 -S sample.ll -o sample_opt.ll

# 5. Generate target assembly from IR
llc sample_opt.ll -o sample.s

# 6. Execute the bitcode directly with the LLVM interpreter/JIT
lli sample.bc

# 7. Or build a native executable from IR and run it
clang sample.ll -o sample
./sample
```

The `-Xclang -disable-O0-optnone` flag is only there so that `opt` can still optimize the `-O0` output; without it `opt` skips functions marked `optnone`.

### 10. Result
The C program was translated into LLVM IR, converted between text and bitcode, optimized, lowered to x86-64 assembly, and executed with both `lli` and a native executable. All produced `Sum = 8`, so the LLVM pipeline Source code -> IR -> Optimization -> Machine code was studied and verified.

## Lab Exercise 8: Type Checking using YACC

**Files:** `typecheck.y`, `typecheck.l`, `input.txt`

**`typecheck.y`:**

```yacc
%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX 100

typedef struct
{
    char name[20];
    char type[10];
} Symbol;

Symbol symbolTable[MAX];
int symbolCount = 0;

void addSymbol(char *name, char *type)
{
    strcpy(symbolTable[symbolCount].name, name);
    strcpy(symbolTable[symbolCount].type, type);
    symbolCount++;
}

char *getType(char *name)
{
    int i;
    for (i = 0; i < symbolCount; i++)
    {
        if (strcmp(symbolTable[i].name, name) == 0)
            return symbolTable[i].type;
    }
    return "undefined";
}

void checkType(char *var, char *exprType)
{
    char *varType = getType(var);

    if (strcmp(varType, "undefined") == 0)
    {
        printf("Error: Variable %s is not declared\n", var);
        return;
    }

    if (strcmp(varType, exprType) == 0)
        printf("Type Checking Successful: %s = %s\n", var, exprType);
    else
        printf("Type Error: Cannot assign %s to %s\n", exprType, varType);
}

int yylex();
void yyerror(char *s);
%}

%union
{
    char str[20];
}

%token INT FLOAT CHAR
%token <str> ID
%token INTEGER REAL CHARACTER
%token ASSIGN SEMICOLON

%type <str> type
%type <str> expression

%%

program:
        declarations assignments
        ;

declarations:
        declarations declaration
        | declaration
        ;

declaration:
        type ID SEMICOLON
        {
            addSymbol($2, $1);
        }
        ;

type:
        INT     { strcpy($$, "int"); }
        | FLOAT { strcpy($$, "float"); }
        | CHAR  { strcpy($$, "char"); }
        ;

assignments:
        assignments assignment
        | assignment
        ;

assignment:
        ID ASSIGN expression SEMICOLON
        {
            checkType($1, $3);
        }
        ;

expression:
        INTEGER     { strcpy($$, "int"); }
        | REAL      { strcpy($$, "float"); }
        | CHARACTER { strcpy($$, "char"); }
        ;

%%

void yyerror(char *s)
{
    printf("Syntax Error: %s\n", s);
}

    return 0;
}
```

## Lab Exercise 9: Balancing Symbols using YACC

**Files:** `balance.y`, `balance.l`

**`balance.y`:**

```yacc
%{
#include <stdio.h>
#include <stdlib.h>

int yylex();
void yyerror(char *s);

char stack[100];
int top = -1;

void push(char c)
{
    stack[++top] = c;
}

char pop()
{
    if (top == -1)
        return '\0';
    return stack[top--];
}

int match(char open, char close)
{
    if (open == '(' && close == ')') return 1;
    if (open == '[' && close == ']') return 1;
    if (open == '{' && close == '}') return 1;
    return 0;
}
%}

%token OPEN_PAREN CLOSE_PAREN
%token OPEN_BRACKET CLOSE_BRACKET
%token OPEN_BRACE CLOSE_BRACE
%token OTHER

%%

input:
        symbols
        {
            if (top == -1)
                printf("Balanced Symbols\n");
            else
                printf("Unbalanced Symbols\n");
        }
        ;

symbols:
        symbols symbol
        | symbol
        ;

symbol:
        OPEN_PAREN    { push('('); }
        | CLOSE_PAREN
        {
            char c = pop();
            if (!match(c, ')')) { yyerror("Unbalanced Symbols"); YYABORT; }
        }
        | OPEN_BRACKET { push('['); }
        | CLOSE_BRACKET
        {
            char c = pop();
            if (!match(c, ']')) { yyerror("Unbalanced Symbols"); YYABORT; }
        }
        | OPEN_BRACE   { push('{'); }
        | CLOSE_BRACE
        {
            char c = pop();
            if (!match(c, '}')) { yyerror("Unbalanced Symbols"); YYABORT; }
        }
        | OTHER
        ;

%%

void yyerror(char *s)
{
    printf("%s\n", s);
}

int main()
{
    printf("Enter an expression:\n");
    yyparse();
    return 0;
}
```

## Quick troubleshooting
- `flex: command not found` -> `sudo apt install flex bison`.
- `undefined reference to yylex/yywrap` -> compile with `-lfl`, or keep the `yywrap()` function that each program already defines.
- `y.tab.h: No such file` -> use `typecheck.tab.h` / `balance.tab.h` (bison naming) as in the provided files.
- On Windows, run everything inside WSL (Ubuntu), not in cmd/PowerShell.
